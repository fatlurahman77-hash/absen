Kode Backend (Code.gs)

Salin kode ini ke Google Apps Script di dalam Spreadsheet DATABASE TK untuk menerima data presensi dan membaca otomatis tab MASTER_GURU dan MASTER_SISWA.

/* =========================================================================
   BACKEND PRODUCTION: DATABASE TK (TK PGRI PAKANDANGAN SANGRA)
   Sheet File : DATABASE TK
   Tabs       : "ABSENSI SISWA", "ABSENSI GURU", "MASTER_GURU", "MASTER_SISWA"
   ========================================================================= */

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = e && e.parameter ? e.parameter.action : "";

    if (action === "GET_MASTER" || action === "GET_GURU") {
      return ContentService.createTextOutput(JSON.stringify(getMasterRecords(ss)))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var sheetSiswa = ss.getSheetByName("ABSENSI SISWA") || ss.getSheetByName("ABSENSI_SISWA");
    var sheetGuru = ss.getSheetByName("ABSENSI GURU") || ss.getSheetByName("ABSENSI_GURU");
    var sheetMasterGuru = ss.getSheetByName("MASTER_GURU") || ss.getSheetByName("MASTER GURU");
    var sheetMasterSiswa = ss.getSheetByName("MASTER_SISWA") || ss.getSheetByName("MASTER SISWA");

    var response = {
      status: "ONLINE",
      service: "Backend Presensi DATABASE TK",
      sekolah: "TK PGRI Pakandangan Sangra",
      spreadsheetName: ss.getName(),
      tabsReady: {
        absensiSiswa: sheetSiswa !== null,
        absensiGuru: sheetGuru !== null,
        masterGuru: sheetMasterGuru !== null,
        masterSiswa: sheetMasterSiswa !== null
      },
      serverTime: Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd HH:mm:ss") + " WIB"
    };

    return ContentService.createTextOutput(JSON.stringify(response))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "ERROR",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  
  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("Payload kosong atau format salah");
    }

    var rawText = e.postData.contents;
    var data = JSON.parse(rawText);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (data.action === "GET_MASTER") {
      return ContentService.createTextOutput(JSON.stringify(getMasterRecords(ss)))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "PING") {
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: "Koneksi DATABASE TK Aktif & Siap Menerima Presensi!"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "ABSENSI_SISWA") {
      var sheetSiswa = ss.getSheetByName("ABSENSI SISWA") || ss.getSheetByName("ABSENSI_SISWA");
      if (!sheetSiswa) {
        sheetSiswa = ss.insertSheet("ABSENSI SISWA");
        sheetSiswa.appendRow(["TIMESTAMP", "ID_ABSENSI_SISWA", "TANGGAL", "NIS", "NAMA_SISWA", "KELOMPOK", "STATUS", "KETERANGAN"]);
      }

      var rowSiswa = [
        data.TIMESTAMP || new Date(),
        data.ID_ABSENSI_SISWA,
        data.TANGGAL,
        "'" + data.NIS,
        data.NAMA_SISWA,
        data.KELOMPOK,
        data.STATUS,
        data.KETERANGAN
      ];

      sheetSiswa.appendRow(rowSiswa);
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: "Presensi Siswa " + data.NAMA_SISWA + " Berhasil Tercatat",
        id: data.ID_ABSENSI_SISWA
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "ABSENSI_GURU") {
      var sheetGuru = ss.getSheetByName("ABSENSI GURU") || ss.getSheetByName("ABSENSI_GURU");
      if (!sheetGuru) {
        sheetGuru = ss.insertSheet("ABSENSI GURU");
        sheetGuru.appendRow(["TIMESTAMP", "ID_ABSENSI_GURU", "TANGGAL", "NUG", "NAMA_GURU", "STATUS", "KETERANGAN", "JAM_DATANG", "JAM_PULANG", "LINK_BUKTI_ABSEN"]);
      }

      var driveUrl = "-";
      if (data.BASE64_IMAGE) {
        driveUrl = uploadSelfieToDrive(data.BASE64_IMAGE, data.NUG, data.NAMA_GURU);
      }

      var rowGuru = [
        data.TIMESTAMP || new Date(),
        data.ID_ABSENSI_GURU,
        data.TANGGAL,
        data.NUG,
        data.NAMA_GURU,
        data.STATUS,
        data.KETERANGAN,
        data.JAM_DATANG || "-",
        data.JAM_PULANG || "-",
        driveUrl
      ];

      sheetGuru.appendRow(rowGuru);
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: "Presensi Guru " + data.NAMA_GURU + " Berhasil Tercatat",
        link: driveUrl
      })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Aksi tidak dikenal" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getMasterRecords(ss) {
  var sheetGuru = ss.getSheetByName("MASTER_GURU") || ss.getSheetByName("MASTER GURU") || ss.getSheetByName("GURU");
  var sheetSiswa = ss.getSheetByName("MASTER_SISWA") || ss.getSheetByName("MASTER SISWA") || ss.getSheetByName("SISWA");

  var guruList = [];
  if (sheetGuru) {
    var valsG = sheetGuru.getDataRange().getValues();
    if (valsG.length > 1) {
      var headG = valsG[0].map(function(h){ return String(h).trim().toUpperCase(); });
      var idxNug = headG.indexOf("NUG");
      if (idxNug === -1) idxNug = 0;
      var idxNamaG = headG.indexOf("NAMA_GURU") !== -1 ? headG.indexOf("NAMA_GURU") : headG.indexOf("NAMA");
      if (idxNamaG === -1) idxNamaG = 1;
      var idxJabatan = headG.indexOf("JABATAN");

      for (var i = 1; i < valsG.length; i++) {
        var rG = valsG[i];
        if (rG[idxNug] || rG[idxNamaG]) {
          guruList.push({
            nug: String(rG[idxNug] || "").trim(),
            nama: String(rG[idxNamaG] || "").trim(),
            jabatan: idxJabatan !== -1 ? String(rG[idxJabatan] || "").trim() : ""
          });
        }
      }
    }
  }

  var siswaObj = {};
  if (sheetSiswa) {
    var valsS = sheetSiswa.getDataRange().getValues();
    if (valsS.length > 1) {
      var headS = valsS[0].map(function(h){ return String(h).trim().toUpperCase(); });
      var idxNis = headS.indexOf("NIS");
      if (idxNis === -1) idxNis = 0;
      var idxNamaS = headS.indexOf("NAMA_SISWA") !== -1 ? headS.indexOf("NAMA_SISWA") : headS.indexOf("NAMA");
      if (idxNamaS === -1) idxNamaS = 1;
      var idxKel = headS.indexOf("KELOMPOK");
      if (idxKel === -1) idxKel = 2;

      for (var j = 1; j < valsS.length; j++) {
        var rS = valsS[j];
        var nisStr = String(rS[idxNis] || "").trim();
        if (nisStr) {
          siswaObj[nisStr] = {
            nama: String(rS[idxNamaS] || "").trim(),
            kelompok: idxKel !== -1 ? String(rS[idxKel] || "").trim() : "TK"
          };
        }
      }
    }
  }

  return {
    success: true,
    guru: guruList,
    siswa: siswaObj
  };
}

function uploadSelfieToDrive(base64Data, nug, nama) {
  try {
    var folderName = "BUKTI_ABSENSI_GURU_TK";
    var folders = DriveApp.getFoldersByName(folderName);
    var targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    var parts = base64Data.split(",");
    var mime = parts[0].match(/:(.*?);/)[1];
    var bytes = Utilities.base64Decode(parts[1]);
    var timeStamp = Utilities.formatDate(new Date(), "GMT+7", "yyyyMMdd_HHmmss");
    var fileName = "SELFIE_" + (nug || "GURU") + "_" + timeStamp + ".jpg";

    var blob = Utilities.newBlob(bytes, mime, fileName);
    var file = targetFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (e) {
    return "ERR_UPLOAD: " + e.message;
  }
}
